# Zapteach Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir a extensão Chrome (MV3) "Zapteach" que insere o nome do atendente na frente de cada mensagem enviada no WhatsApp Web.

**Architecture:** Content script intercepta o Enter no campo de digitação do WhatsApp Web, monta a mensagem final (nome formatado + texto) usando funções puras e dispara o envio. Um popup edita a configuração, salva em `chrome.storage.sync` (sincroniza pela conta Google). A lógica de montagem fica isolada e testável sem navegador.

**Tech Stack:** Chrome Extension Manifest V3, JavaScript puro (sem dependências de runtime), testes com `node --test` (embutido no Node 22).

## Global Constraints

- Manifest V3, `manifest_version: 3`.
- Sem dependências de terceiros em runtime (JS puro). Testes usam apenas `node:test` e `node:assert`.
- Nome do produto: **Zapteach**. Versão inicial: `1.0.0`.
- Formatação usa markdown nativo do WhatsApp: negrito `*...*`, itálico `_..._`.
- Config sincroniza via `chrome.storage.sync`; nunca `localStorage`.
- "Frase inicial maiúscula" aplicada **no envio** (não ao vivo).
- Schema da config (exato):
  `{ nome: string, ativo: bool, negrito: bool, italico: bool, quebraLinha: bool, fraseMaiuscula: bool }`.
- Todos os arquivos de código em `src/`; testes em `test/`.
- `format.js` deve funcionar tanto como content script (funções no escopo compartilhado) quanto via `require()` no Node (exporta com guarda `typeof module`).

```js
// Lógica pura de montagem da mensagem. Sem dependência de navegador.
// Funciona como content script (escopo compartilhado) e via require() no Node.

const DEFAULT_CONFIG = {
  nome: "",
  ativo: true,
  negrito: false,
  italico: false,
  quebraLinha: true,
  fraseMaiuscula: true,
};

function formatarNome(config) {
  const nome = (config.nome || "").trim();
  if (!nome) return "";
  let resultado = nome;
  if (config.italico) resultado = `_${resultado}_`;
  if (config.negrito) resultado = `*${resultado}*`;
  return resultado;
}

function capitalizarFrases(texto) {
  return texto.replace(
    /(^\s*|[.!?]\s+|\n\s*)([a-zà-ÿ])/g,
    (_, prefixo, letra) => prefixo + letra.toUpperCase()
  );
}

function montarMensagem(config, texto) {
  const corpo = config.fraseMaiuscula ? capitalizarFrases(texto) : texto;
  const prefixo = formatarNome(config);
  if (!prefixo) return corpo;
  const separador = config.quebraLinha ? "\n" : " ";
  return prefixo + separador + corpo;
}

if (typeof module !== "undefined" && module.exports) {
  module.exports = { DEFAULT_CONFIG, formatarNome, capitalizarFrases, montarMensagem };
}
```

```js
function carregar() {
  chrome.storage.sync.get(DEFAULT_CONFIG, (cfg) => {
    document.getElementById("nome").value = cfg.nome;
    document.getElementById("ativo").checked = cfg.ativo;
    document.getElementById("negrito").checked = cfg.negrito;
    document.getElementById("italico").checked = cfg.italico;
    document.getElementById("quebraLinha").checked = cfg.quebraLinha;
    document.getElementById("fraseMaiuscula").checked = cfg.fraseMaiuscula;
  });
}

function salvar() {
  const cfg = {
    nome: document.getElementById("nome").value.trim(),
    ativo: document.getElementById("ativo").checked,
    negrito: document.getElementById("negrito").checked,
    italico: document.getElementById("italico").checked,
    quebraLinha: document.getElementById("quebraLinha").checked,
    fraseMaiuscula: document.getElementById("fraseMaiuscula").checked,
  };
  chrome.storage.sync.set(cfg, () => {
    const status = document.getElementById("status");
    status.textContent = "Salvo!";
    setTimeout(() => (status.textContent = ""), 1500);
  });
}

document.addEventListener("DOMContentLoaded", () => {
  carregar();
  document.getElementById("salvar").addEventListener("click", salvar);
});
```

- [ ] **Step 1: Criar `src/lib/seletores.js`**

```js
// Seletores do DOM do WhatsApp Web centralizados.
// Se o WhatsApp mudar o layout e a injeção parar de funcionar, ajuste AQUI.
const SELETORES = {
  caixaTexto: 'footer div[contenteditable="true"][role="textbox"]',
  botaoEnviar: '[data-icon="send"], [data-icon="wds-ic-send-filled"]',
};
```

- [ ] **Step 2: Criar `src/content.js`**

```js
// Roda dentro do WhatsApp Web. Carregado após format.js e seletores.js
// (mesmo escopo isolado), então DEFAULT_CONFIG, montarMensagem e SELETORES
// já estão disponíveis.

let configAtual = { ...DEFAULT_CONFIG };

chrome.storage.sync.get(DEFAULT_CONFIG, (dados) => {
  configAtual = { ...DEFAULT_CONFIG, ...dados };
});

chrome.storage.onChanged.addListener((mudancas, area) => {
  if (area !== "sync") return;
  for (const chave of Object.keys(mudancas)) {
    if (chave in configAtual) configAtual[chave] = mudancas[chave].newValue;
  }
});

function obterCaixaDeTexto(alvo) {
  if (alvo && alvo.matches && alvo.matches(SELETORES.caixaTexto)) return alvo;
  return document.querySelector(SELETORES.caixaTexto);
}

function enviarMensagem() {
  const icone = document.querySelector(SELETORES.botaoEnviar);
  const botao = icone && icone.closest("button");
  if (botao) botao.click();
  else if (icone) icone.click();
}

document.addEventListener(
  "keydown",
  (e) => {
    if (e.key !== "Enter" || e.shiftKey || e.isComposing) return;
    if (!configAtual.ativo) return;

    const caixa = obterCaixaDeTexto(e.target);
    if (!caixa) return;

    const texto = caixa.textContent ?? "";
    if (!texto.trim()) return;

    const final = montarMensagem(configAtual, texto);
    if (final === texto) return; // nada a acrescentar

    // Impede o envio original e reescreve com o nome antes de enviar.
    e.preventDefault();
    e.stopPropagation();

    caixa.focus();
    document.execCommand("selectAll", false, null);
    document.execCommand("insertText", false, final);

    // Dá um tick para o WhatsApp registrar o texto e exibir o botão enviar.
    setTimeout(enviarMensagem, 0);
  },
  true // capture: intercepta antes do handler do WhatsApp
);
```
